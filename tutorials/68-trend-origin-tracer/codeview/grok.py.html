74class GrokError(RuntimeError):
75 pass

78def search(prompt: str, tool: dict, *, max_turns: int, api_key: str | None = None, retries: int = 3) -> dict:
79 body = {
80 "model": MODEL,
81 "input": [{"role": "user", "content": prompt}],
82 "tools": [{"type": "x_search", **tool}],
83 "max_turns": max_turns,
84 "text": {"format": {"type": "json_schema", "name": "posts", "schema": POST_SCHEMA, "strict": True}},
85 }
86 headers = {"Authorization": f"Bearer {api_key or os.environ['XAI_API_KEY']}"}
87 last = None
88 for attempt in range(retries):
89 try:
90 resp = httpx.post(XAI_URL, headers=headers, json=body, timeout=600)
91 except httpx.HTTPError as exc:
92 last = str(exc)
93 else:
94 if resp.status_code == 200:
95 return parse_response(resp.json())
96 last = f"HTTP {resp.status_code}: {resp.text[:300]}"
97 if resp.status_code not in (408, 429, 500, 502, 503, 504):
98 break
99 time.sleep(2 ** attempt * 3)
100 raise GrokError(last or "unknown error")

103def parse_response(data: dict) -> dict:
104 posts, citations, calls = [], list(data.get("citations") or []), []
105 for item in data.get("output", []):
106 if item.get("type") == "custom_tool_call":
107 calls.append({"name": item.get("name"), "input": item.get("input")})
108 if item.get("type") != "message":
109 continue
110 for content in item.get("content", []):
111 if content.get("type") != "output_text":
112 continue
113 citations += [a.get("url", "") for a in content.get("annotations", []) if a.get("type") == "url_citation"]
114 try:
115 posts += json.loads(content.get("text") or "{}").get("posts", [])
116 except json.JSONDecodeError:
117 pass
118 usage = data.get("usage") or {}
119 return {
120 "posts": posts,
121 "citations": citations,
122 "tool_calls": calls,
123 "cost_usd": (usage.get("cost_in_usd_ticks") or 0) / 1e10,
124 "posts_fetched": (usage.get("server_side_tool_usage_details") or {}).get("x_posts_fetched", 0),
125 }
