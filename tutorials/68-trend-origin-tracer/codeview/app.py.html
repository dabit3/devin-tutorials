37def normalize(phrase: str) -> str:
38 return re.sub(r"\s+", " ", (phrase or "").strip().strip('"').strip())

41def slug(phrase: str) -> str:
42 text = normalize(phrase).lower()
43 key = re.sub(r"[^0-9a-z]+", "-", text).strip("-")
44 if key != text.replace(" ", "-"):
45 key += "-" + hashlib.sha1(text.encode()).hexdigest()[:8]
46 return key

59@app.function(secrets=[xai_secret], timeout=600, max_containers=24)
60def run_search(task: dict) -> dict:
61 """One Grok x_search call. Verification happens here, where the citations are available."""
62 from tracer import grok
63 from tracer.graph import verify_posts
64
65 kind, phrase = task["kind"], task["phrase"]
66 tool, label = {}, kind
67 if kind in ("earliest", "recent"):
68 if task.get("from"):
69 tool["from_date"] = task["from"]
70 if task.get("to"):
71 tool["to_date"] = task["to"]
72 label = f"{kind} {task.get('from', 'start')} → {task.get('to', 'now')}"
73 prompt = grok.earliest_prompt(phrase) if kind == "earliest" else grok.recent_prompt(phrase)
74 max_turns = 3 if kind == "earliest" else 2
75 else:
76 prompt, max_turns, label = grok.spread_prompt(task["post_id"]), 2, f"spread of {task['post_id']}"
77
78 started = time.time()
79 try:
80 result = grok.search(prompt, tool, max_turns=max_turns)
81 except Exception as exc: # noqa: BLE001 - one failed window should not sink the trace
82 return {"label": label, "error": str(exc)[:300], "posts": [], "dropped": 0, "cost_usd": 0, "seconds": 0}
83 posts, dropped = verify_posts(result["posts"], result["citations"], phrase, label)
84 return {
85 "label": label,
86 "posts": [p.to_json() for p in posts],
87 "dropped": dropped,
88 "cost_usd": round(result["cost_usd"], 4),
89 "x_posts_fetched": result["posts_fetched"],
90 "queries": [c["input"] for c in result["tool_calls"]],
91 "seconds": round(time.time() - started, 1),
92 }

95def _fan_out(tasks: list[dict], log: list[dict]):
96 from tracer.graph import Post
97
98 batches = []
99 for res in run_search.map(tasks, return_exceptions=True):
100 if isinstance(res, Exception):
101 log.append({"label": "search", "error": str(res)[:300]})
102 continue
103 log.append({k: v for k, v in res.items() if k != "posts"} | {"kept": len(res["posts"])})
104 batches.append([Post.from_json(p) for p in res["posts"]])
105 return batches

108@app.function(timeout=2400)
109def run_trace(phrase: str, mode: str = "full") -> dict:
110 from tracer.graph import Post, build_trace, merge_posts
111 from tracer.plan import coarse_windows, split_window, window_containing
112
113 phrase = normalize(phrase)
114 key = slug(phrase)
115 previous = store.get(f"trace:{key}")
116 if mode == "refresh" and not previous:
117 mode = "full"
118 log: list[dict] = []
119 started = time.time()
120 today = datetime.now(timezone.utc).date()
121 task = lambda kind, **kw: {"kind": kind, "phrase": phrase, **kw} # noqa: E731
122 try:
123 if mode == "full":
124 windows = coarse_windows(today)
125 set_status(phrase, "running", f"Searching {len(windows) + 1} time windows on X in parallel", started_at=started)
126 tasks = [task("earliest", **{"from": a.isoformat(), "to": b.isoformat()}) for a, b in windows]
127 tasks.append(task("recent"))
128 posts = merge_posts(*_fan_out(tasks, log))
129
130 for round_no in range(REFINE_ROUNDS):
131 matching = sorted((p for p in posts.values() if p.matches_phrase), key=lambda p: p.time)
132 if not matching:
133 break
134 earliest = matching[0]
135 win = window_containing(windows, earliest.time)
136 if not win or (earliest.time.date() - win[0]).days <= 7:
137 break
138 windows = split_window(win[0], earliest.time.date() + timedelta(days=1), REFINE_PARTS)
139 set_status(phrase, "running", f"Narrowing in on the first post (round {round_no + 1}): "
140 f"{len(windows)} windows {win[0]} → {earliest.time.date()}")
141 tasks = [task("earliest", **{"from": a.isoformat(), "to": b.isoformat()}) for a, b in windows]
142 posts = merge_posts(list(posts.values()), *_fan_out(tasks, log))
143 if min((p.time for p in posts.values() if p.matches_phrase), default=earliest.time) >= earliest.time:
144 break
145 else:
146 posts = merge_posts([Post.from_json(p) for p in previous["posts"]])
147 since = date.fromisoformat(previous["updated_at"][:10]) - timedelta(days=1)
148 set_status(phrase, "running", f"Daily refresh: searching X since {since}", started_at=started)
149 tasks = [task("recent", **{"from": since.isoformat(), "to": (today + timedelta(days=1)).isoformat()}),
150 task("recent")]
151 posts = merge_posts(list(posts.values()), *_fan_out(tasks, log))
152
153 matching = sorted((p for p in posts.values() if p.matches_phrase), key=lambda p: p.time)
154 if matching:
155 seeds = [matching[0]] + sorted(matching[1:], key=lambda p: p.engagement, reverse=True)
156 if mode == "refresh":
157 seeds = [p for p in seeds if "spread" not in " ".join(p.sources)][:3] or seeds[:1]
158 seeds = seeds[: SPREAD_SEEDS + 1]
159 set_status(phrase, "running", f"Following replies and quotes of {len(seeds)} key posts in parallel")
160 posts = merge_posts(list(posts.values()), *_fan_out([task("spread", post_id=p.id) for p in seeds], log))
161 for seed in seeds:
162 posts[seed.id].sources.add(f"spread of {seed.id}")
163
164 searches = (previous or {}).get("stats", {}).get("searches", []) if mode == "refresh" else []
165 searches = (searches + log)[-80:]
166 trace = build_trace(phrase, posts, {
167 "searches": searches,
168 "cost_usd": round(sum(s.get("cost_usd", 0) for s in searches), 3),
169 "dropped_uncited": sum(s.get("dropped", 0) for s in searches),
170 "seconds": round(time.time() - started, 1),
171 })
172 trace.update(updated_at=now_iso(), created_at=(previous or {}).get("created_at", now_iso()), mode=mode)
173 store[f"trace:{key}"] = trace
174 set_status(phrase, "done", "Done", finished_at=time.time())
175 return trace["stats"]
176 except Exception as exc:
177 set_status(phrase, "error", f"Trace failed: {str(exc)[:200]}")
178 raise

185def _start(phrase: str, mode: str = "full") -> dict:
186 key = slug(phrase)
187 status = store.get(f"status:{key}") or {}
188 if status.get("state") == "running" and time.time() - status.get("updated_at", 0) < STALE_RUN_SECONDS:
189 return status
190 if mode == "full":
191 day = date.today().isoformat()
192 if not any(store.put(f"budget:{day}:{i}", key, skip_if_exists=True) for i in range(MAX_TRACES_PER_DAY)):
193 return {"state": "error", "step": f"Daily limit of {MAX_TRACES_PER_DAY} new traces reached; try tomorrow."}
194 store[f"status:{key}"] = {"state": "running", "step": "Queued", "updated_at": time.time(), "started_at": time.time()}
195 run_trace.spawn(phrase, mode)
196 return store[f"status:{key}"]

199@app.function(schedule=modal.Cron("0 6 * * *") if DAILY_CRON else None, timeout=600)
200def refresh_saved() -> list[str]:
201 store["last_refresh"] = time.time()
202 started = []
203 for phrase in _saved():
204 status = _start(phrase, "refresh" if store.get(f"trace:{slug(phrase)}") else "full")
205 if status.get("step") == "Queued":
206 started.append(phrase)
207 return started

219 def clean(phrase: str) -> str:
220 phrase = normalize(phrase)
221 if not 2 <= len(phrase) <= 80 or not slug(phrase):
222 raise HTTPException(400, "Phrase must be 2-80 characters.")
223 if not re.fullmatch(r"[0-9A-Za-z' _-]+", phrase):
224 raise HTTPException(400, "Use letters, numbers, spaces, hyphens, underscores or apostrophes.")
225 return phrase
226
227 def refresh_if_due() -> None:
228 if time.time() - (store.get("last_refresh") or 0) > REFRESH_EVERY_SECONDS:
229 store["last_refresh"] = time.time()
230 refresh_saved.spawn()

232 def summary(phrase: str) -> dict:
233 trace = store.get(f"trace:{slug(phrase)}") or {}
234 status = store.get(f"status:{slug(phrase)}") or {}
235 origin = trace.get("origin") or {}
236 return {"phrase": phrase, "updated_at": trace.get("updated_at"), "state": status.get("state"),
237 "origin_author": origin.get("author"), "origin_time": origin.get("time"),
238 "posts": trace.get("stats", {}).get("phrase_posts")}

250 @api.post("/api/trace")
251 def post_trace(req: dict = Body(...)):
252 phrase = clean(str(req.get("phrase") or ""))
253 if req.get("save"):
254 add_saved(phrase)
255 trace = store.get(f"trace:{slug(phrase)}")
256 fresh = trace and datetime.fromisoformat(trace["updated_at"]) > datetime.now(timezone.utc) - timedelta(hours=24)
257 status = store.get(f"status:{slug(phrase)}") if fresh else _start(phrase, "full")
258 return get_trace(phrase) | {"status": status}

265 @api.post("/api/saved")
266 def add_saved(phrase: str):
267 phrase = clean(phrase)
268 saved = _saved()
269 if phrase.lower() not in [p.lower() for p in saved]:
270 if len(saved) >= MAX_SAVED:
271 raise HTTPException(400, f"At most {MAX_SAVED} saved phrases.")
272 store["saved"] = saved + [phrase]
273 return list_saved()
